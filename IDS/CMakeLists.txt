cmake_minimum_required(VERSION 3.16)
project(canids LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(canids SHARED src/canids.c)
target_include_directories(canids PUBLIC include)
target_compile_options(canids PRIVATE -Wall -Wextra -Wpedantic -Werror)
include(CTest)
if(BUILD_TESTING)
    add_executable(test_canids tests/test_canids.c)
    target_link_libraries(test_canids PRIVATE canids)
    target_compile_options(test_canids PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME canids_native COMMAND test_canids)
endif()
