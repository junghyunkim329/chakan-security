cmake_minimum_required(VERSION 3.16)
project(canids LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(canids_core OBJECT src/canids.c)
set_target_properties(canids_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(canids_core PUBLIC include)
target_compile_options(canids_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_library(canids SHARED $<TARGET_OBJECTS:canids_core>)
target_include_directories(canids PUBLIC include)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONC REQUIRED IMPORTED_TARGET json-c)
pkg_check_modules(ARCHIVE REQUIRED IMPORTED_TARGET libarchive)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
set(CANIDS_RUNTIME_SOURCES src/main.c src/runtime_policy.c src/runtime_input.c src/runtime_log.c)
set(build_sources "{")
foreach(source IN ITEMS ${CANIDS_RUNTIME_SOURCES} src/runtime.h src/canids.c include/canids.h CMakeLists.txt)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${source}")
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${source}" digest)
    string(APPEND build_sources "\"${source}\":\"${digest}\",")
endforeach()
string(REGEX REPLACE ",$" "}" build_sources "${build_sources}")
string(REPLACE "\"" "\\\"" build_sources "${build_sources}")
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
    OUTPUT_VARIABLE build_commit OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/canids_build.h"
    "#define CANIDS_BUILD_COMMIT \"${build_commit}\"\n#define CANIDS_BUILD_SOURCES \"${build_sources}\"\n")
add_executable(canids_cli ${CANIDS_RUNTIME_SOURCES} $<TARGET_OBJECTS:canids_core>)
set_target_properties(canids_cli PROPERTIES OUTPUT_NAME canids)
target_include_directories(canids_cli PRIVATE include "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(canids_cli PRIVATE PkgConfig::JSONC PkgConfig::ARCHIVE OpenSSL::Crypto)
target_compile_options(canids_cli PRIVATE -Wall -Wextra -Wpedantic -Werror)
include(CTest)
if(BUILD_TESTING)
    add_executable(test_canids tests/test_canids.c)
    target_link_libraries(test_canids PRIVATE canids)
    target_compile_options(test_canids PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME canids_native COMMAND test_canids)
endif()
